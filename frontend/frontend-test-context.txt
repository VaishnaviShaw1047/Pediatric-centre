import { HttpClient } from '@angular/common/http';
import { Injectable, computed, inject, signal } from '@angular/core';
import { Router } from '@angular/router';
import { map } from 'rxjs';
import { API_URL } from './config';
import {
  AuthTokenResponse,
  DoctorTokenResponse,
  RegistrationRequest,
  RegistrationResponse,
  Role,
} from './models';

export interface Session {
  token: string;
  role: Role;
  userId: number;
  name?: string;
  mustChangePassword?: boolean;
}

const STORAGE_KEY = 'pc_session';

@Injectable({ providedIn: 'root' })
export class AuthService {
  private http = inject(HttpClient);
  private router = inject(Router);

  private _session = signal<Session | null>(this.restore());
  readonly session = this._session.asReadonly();
  readonly role = computed(() => this._session()?.role ?? null);

  get token(): string | null {
    return this._session()?.token ?? null;
  }

  loginGuardian(mobile: string, password: string) {
    return this.http
      .post<AuthTokenResponse>(`${API_URL}/auth/login`, { mobile_number: mobile, password })
      .pipe(map((r) => this.start(r.access_token, 'guardian')));
  }

  loginDoctor(staffId: string, password: string) {
    return this.http
      .post<DoctorTokenResponse>(`${API_URL}/auth/doctor-login`, { staff_id: staffId, password })
      .pipe(
        map((r) =>
          this.start(r.access_token, 'doctor', {
            name: r.doctor_name,
            mustChangePassword: r.must_change_password,
          }),
        ),
      );
  }

 loginAdmin(username: string, password: string) {
  debugger;

  return this.http
    .post<AuthTokenResponse>(
      `${API_URL}/auth/admin-login`,
      { username, password }
    )
    .pipe(
      map((r) => {
        debugger;
        console.log('Admin login response:', r);
        console.log('Access token:', r.access_token);
        return this.start(r.access_token, 'admin');
      })
    );
}

    register(payload: RegistrationRequest) {
    return this.http.post<RegistrationResponse>(`${API_URL}/registration`, payload);
  }
    markPasswordChanged() {
    const s = this._session();
    if (!s) return;
    const updated = { ...s, mustChangePassword: false };
    localStorage.setItem(STORAGE_KEY, JSON.stringify(updated));
    this._session.set(updated);
  }

  logout() {
    localStorage.removeItem(STORAGE_KEY);
    this._session.set(null);
    this.router.navigate(['/login']);
  }

  private start(token: string, role: Role, extra: Partial<Session> = {}): Session {
    const claims = this.decode(token);
    const session: Session = { token, role, userId: Number(claims?.sub), ...extra };
    localStorage.setItem(STORAGE_KEY, JSON.stringify(session));
    this._session.set(session);
    return session;
  }

  private restore(): Session | null {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (!raw) return null;
      const session: Session = JSON.parse(raw);
      const exp = this.decode(session.token)?.exp;
      if (exp && exp * 1000 < Date.now()) {
        localStorage.removeItem(STORAGE_KEY);
        return null;
      }
      return session;
    } catch {
      return null;
    }
  }

  private decode(token: string): { sub?: string; exp?: number } | null {
    try {
      const payload = token.split('.')[1].replace(/-/g, '+').replace(/_/g, '/');
      return JSON.parse(atob(payload));
    } catch {
      return null;
    }
  }
}



========== src\app\core\role.guard.ts ==========

import { inject } from '@angular/core';
import { CanActivateFn, Router } from '@angular/router';
import { AuthService } from './auth.service';
import { Role } from './models';

export const roleGuard =
  (role: Role): CanActivateFn =>
  () => {
    const auth = inject(AuthService);
    const router = inject(Router);
    return auth.role() === role ? true : router.createUrlTree(['/login']);
  };
  export const passwordChangedGuard: CanActivateFn = () => {
  const auth = inject(AuthService);
  const router = inject(Router);
  return auth.session()?.mustChangePassword
    ? router.createUrlTree(['/doctor/change-password'])
    : true;
};

function underEighteen(control: AbstractControl): ValidationErrors | null {
  const value = control.value;
  if (!value) return null;
  const dob = new Date(value);
  const today = new Date();
  if (dob > today) return { future: true };
  let age = today.getFullYear() - dob.getFullYear();
  const monthDiff = today.getMonth() - dob.getMonth();
  if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < dob.getDate())) age--;
  return age >= 18 ? { adult: true } : null;
}

      // Expired or invalid token on a protected call: sign out.

        auth.logout();
      }

  );

  private restore(): Session | null {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (!raw) return null;
      const session: Session = JSON.parse(raw);
      const exp = this.decode(session.token)?.exp;
      if (exp && exp * 1000 < Date.now()) {
        localStorage.removeItem(STORAGE_KEY);
        return null;
      }
      return session;
    } catch {
      return null;
    }
  }

  private decode(token: string): { sub?: string; exp?: number } | null {
    try {
      const payload = token.split('.')[1].replace(/-/g, '+').replace(/_/g, '/');
      return JSON.parse(atob(payload));
    } catch {
      return null;
    }
  }
}

import { HttpErrorResponse } from '@angular/common/http';

export function errorMessage(err: HttpErrorResponse, fallback: string): string {
  const detail = err.error?.detail;
  if (typeof detail === 'string') return detail;
  if (Array.isArray(detail)) {
    return detail
      .map((d: { loc?: unknown[]; msg: string }) => `${(d.loc ?? []).slice(1).join(' > ')}: ${d.msg}`)
      .join('; ');
  }
  return err.status === 0 ? 'Cannot reach the server. Is the backend running?' : fallback;
}

export class BookAppointment implements OnInit {
  private api = inject(ParentApi);
  private route = inject(ActivatedRoute);

  doctorId = Number(this.route.snapshot.paramMap.get('doctorId'));
  today = new Date(Date.now() - new Date().getTimezoneOffset() * 60000).toISOString().slice(0, 10);

  doctor = signal<DoctorPublic | null>(null);
  children = signal<PatientSummary[]>([]);
  slots = signal<string[]>([]);

  patientId = signal<number | null>(null);
  date = signal('');
  slot = signal<string | null>(null);
  reason = signal('');

  loading = signal(true);
  loadingSlots = signal(false);
  submitting = signal(false);
  error = signal<string | null>(null);
  booked = signal<Appointment | null>(null);

  canBook = computed(() => this.patientId() !== null && !!this.date() && !!this.slot());

  ngOnInit() {
    if (!Number.isInteger(this.doctorId)) {
      this.error.set('Invalid doctor.');
      this.loading.set(false);
      return;
    }

    this.api.doctor(this.doctorId).subscribe({
      next: (d) => this.doctor.set(d),
      error: (e) => this.error.set(this.errorText(e, 'Could not load this doctor.')),
    });

    this.api.myChildren().subscribe({
      next: (rows) => {
        this.children.set(rows);
        if (rows.length === 1) this.patientId.set(rows[0].patient_id);
        this.loading.set(false);
      },
      error: (e) => {
        this.loading.set(false);
        this.error.set(this.errorText(e, 'Could not load your children.'));
      },
    });
  }

  onChild(event: Event) {
    const value = (event.target as HTMLSelectElement).value;
    this.patientId.set(value ? Number(value) : null);
  }

  onDate(event: Event) {
    const value = (event.target as HTMLInputElement).value;
    this.date.set(value);
    this.slot.set(null);
    this.error.set(null);
    if (value) this.loadSlots();
    else this.slots.set([]);
  }

  onReason(event: Event) {
    this.reason.set((event.target as HTMLTextAreaElement).value);
  }

  private loadSlots() {
    this.loadingSlots.set(true);
    this.api.slots(this.doctorId, this.date()).subscribe({
      next: (res) => {
        this.slots.set(res.slots);
        this.loadingSlots.set(false);
      },
      error: (e) => {
        this.slots.set([]);
        this.loadingSlots.set(false);
        this.error.set(this.errorText(e, 'Could not load time slots.'));
      },
    });
  }

  submit() {
    if (!this.canBook()) return;
    this.submitting.set(true);
    this.error.set(null);

    this.api
      .book({
        doctor_id: this.doctorId,
        patient_id: this.patientId()!,
        scheduled_at: `${this.date()}T${this.slot()}:00`,
        reason_for_visit: this.reason().trim() || null,
      })
      .subscribe({
        next: (appointment) => {
          this.submitting.set(false);
          this.booked.set(appointment);
        },
        error: (e) => {
          this.submitting.set(false);
          this.error.set(this.errorText(e, 'Booking failed. Please try again.'));
          // The slot may have just been taken, so refresh the list.
          this.slot.set(null);
          this.loadSlots();
        },
      });
  }

  private errorText(err: HttpErrorResponse, fallback: string): string {
    const detail = err.error?.detail;
    if (typeof detail === 'string') return detail;
    if (Array.isArray(detail)) {
      return detail.map((d: { msg: string }) => d.msg).join('; ');
    }
    return err.status === 0 ? 'Cannot reach the server. Is the backend running?' : fallback;
  }
}

export class FindDoctors implements OnInit {
  private api = inject(ParentApi);
  private fb = inject(NonNullableFormBuilder);

  days = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
  specialties = signal<SpecialtyCount[]>([]);
  doctors = signal<DoctorPublic[]>([]);
  loading = signal(false);
  error = signal<string | null>(null);

  filters = this.fb.group({
    specialty: '',
    language: '',
    max_fee: '',
    day: '',
  });

  ngOnInit() {
    this.api.specialties().subscribe({
      next: (rows) => this.specialties.set(rows),
      error: () => {}, // the filter still works without the dropdown counts
    });
    this.search();
  }

  search() {
    const f = this.filters.getRawValue();
    this.loading.set(true);
    this.error.set(null);

    this.api
      .doctors({
        specialty: f.specialty,
        language: f.language,
        day: f.day,
        max_fee: f.max_fee === '' ? null : Number(f.max_fee),
      })
      .subscribe({
        next: (rows) => {
          this.doctors.set(rows);
          this.loading.set(false);
        },
        error: (err: HttpErrorResponse) => {
          this.loading.set(false);
          this.error.set(
            typeof err.error?.detail === 'string'
              ? err.error.detail
              : 'Could not load doctors. Is the backend running?',
          );
        },
      });
  }

  reset() {
    this.filters.reset();
    this.search();
  }
}

export class MyAppointments implements OnInit {
  private api = inject(ParentApi);

  appointments = signal<Appointment[]>([]);
  upcomingOnly = signal(true);
  loading = signal(false);
  cancelling = signal(false);
  confirming = signal<string | null>(null);
  error = signal<string | null>(null);

  ngOnInit() {
    this.load();
  }

  setTab(upcomingOnly: boolean) {
    this.upcomingOnly.set(upcomingOnly);
    this.confirming.set(null);
    this.load();
  }

  canCancel(a: Appointment): boolean {
    return a.status !== 'cancelled' && a.status !== 'completed';
  }

  load() {
    this.loading.set(true);
    this.error.set(null);
    this.api.appointments(this.upcomingOnly()).subscribe({
      next: (rows) => {
        this.appointments.set(rows);
        this.loading.set(false);
      },
      error: (e) => {
        this.loading.set(false);
        this.error.set(this.errorText(e, 'Could not load appointments.'));
      },
    });
  }

  cancel(appointment: Appointment) {
    this.cancelling.set(true);
    this.error.set(null);
    this.api.cancel(appointment.appointment_ref, null).subscribe({
      next: (updated) => {
        this.cancelling.set(false);
        this.confirming.set(null);
        this.appointments.update((list) =>
          list.map((a) => (a.appointment_ref === updated.appointment_ref ? updated : a)),
        );
      },
      error: (e) => {
        this.cancelling.set(false);
        this.error.set(this.errorText(e, 'Could not cancel this appointment.'));
      },
    });
  }

  private errorText(err: HttpErrorResponse, fallback: string): string {
    const detail = err.error?.detail;
    if (typeof detail === 'string') return detail;
    return err.status === 0 ? 'Cannot reach the server. Is the backend running?' : fallback;
  }
}

export class TreatmentHistory implements OnInit {

  private api = inject(ParentApi);

  children = signal<PatientSummary[]>([]);
  patientId = signal<number | null>(null);
  records = signal<any[]>([]);

  loading = signal(false);
  error = signal<string | null>(null);

  ngOnInit(): void {
    this.api.myChildren().subscribe({
      next: (rows: PatientSummary[]) => {
        this.children.set(rows);

        if (rows.length === 1) {
          this.patientId.set(rows[0].patient_id);
          this.load();
        }
      },

      error: (e: HttpErrorResponse) => {
        this.error.set(
          this.errorText(e, 'Could not load your children.')
        );
      },
    });
  }

  onChild(event: Event): void {
    const value = (event.target as HTMLSelectElement).value;

    this.patientId.set(value ? Number(value) : null);
    this.records.set([]);

    if (value) {
      this.load();
    }
  }

  private load(): void {
    const id = this.patientId();

    if (id === null) {
      return;
    }

    this.loading.set(true);
    this.error.set(null);

    this.api.treatmentHistory(id).subscribe({
      next: (rows: any[]) => {
        this.records.set(rows);
        this.loading.set(false);
      },

      error: (e: HttpErrorResponse) => {
        this.loading.set(false);

        this.error.set(
          this.errorText(e, 'Could not load treatment records.')
        );
      },
    });
  }

  private errorText(
    err: HttpErrorResponse,
    fallback: string
  ): string {
    const detail = err.error?.detail;

    if (typeof detail === 'string') {
      return detail;
    }

    return err.status === 0
      ? 'Cannot reach the server. Is the backend running?'
      : fallback;
  }
}
